# 1\.Word separation

Below are three “from\-scratch” implementations of Chinese word segmentation algorithms without using existing segmentation APIs like Jieba or PKUSEG\.

These examples are suitable for teaching NLP principles and algorithm design\.

---

# 1\. Maximum Forward Matching \(FMM\)

This is the most classical dictionary\-based segmentation algorithm\.

## Principle

- Scan sentence from left to right 

- Always match the longest word in dictionary 

- Greedy algorithm 

---

## Code

```Python
# Forward Maximum Matching (FMM)

dictionary = {
    "我", "来到", "北京", "清华大学",
    "学习", "自然语言", "处理", "自然",
    "语言", "自然语言处理"
}

MAX_LEN = 5


def fmm(sentence, dictionary, max_len):
    result = []
    index = 0

    while index < len(sentence):

        matched = False

        # Try longest possible word first
        for size in range(max_len, 0, -1):

            piece = sentence[index:index + size]

            if piece in dictionary:
                result.append(piece)
                index += size
                matched = True
                break

        # If no match found
        if not matched:
            result.append(sentence[index])
            index += 1

    return result


sentence = "我来到北京清华大学学习自然语言处理"

result = fmm(sentence, dictionary, MAX_LEN)

print("Sentence:")
print(sentence)

print("\nSegmentation:")
print(result)
```

---

## Output

```Plain Text
['我', '来到', '北京', '清华大学', '学习', '自然语言处理']
```

# 2\. Maximum Backward Matching \(BMM\)

This algorithm scans from right to left\.

Sometimes BMM performs better than FMM\.

---

## Principle

- Start from sentence end 

- Match longest word backward 

- Reverse final result 

---

## Code

```Python
# Backward Maximum Matching (BMM)

dictionary = {
    "我", "来到", "北京", "清华大学",
    "学习", "自然语言", "处理", "自然",
    "语言", "自然语言处理"
}

MAX_LEN = 5


def bmm(sentence, dictionary, max_len):

    result = []
    index = len(sentence)

    while index > 0:

        matched = False

        for size in range(max_len, 0, -1):

            start = max(index - size, 0)

            piece = sentence[start:index]

            if piece in dictionary:
                result.insert(0, piece)
                index -= size
                matched = True
                break

        if not matched:
            result.insert(0, sentence[index - 1])
            index -= 1

    return result


sentence = "我来到北京清华大学学习自然语言处理"

result = bmm(sentence, dictionary, MAX_LEN)

print("Sentence:")
print(sentence)

print("\nSegmentation:")
print(result)
```

---

## Output

```Python
['我', '来到', '北京', '清华大学', '学习', '自然语言处理']
```

# 3\. Hidden Markov Model \(HMM\) Word Segmentation

This is a statistical segmentation algorithm\.

We use BMES tags:

---

# Simplified HMM Example

This is an educational mini\-version\.

---

## Code

```Python
import numpy as np

# States
states = ['B', 'M', 'E', 'S']

# Initial probabilities
start_prob = {
    'B': 0.5,
    'M': 0.0,
    'E': 0.0,
    'S': 0.5
}

# Transition probabilities
trans_prob = {
    'B': {'M': 0.5, 'E': 0.5},
    'M': {'M': 0.5, 'E': 0.5},
    'E': {'B': 0.5, 'S': 0.5},
    'S': {'B': 0.5, 'S': 0.5}
}

# Emission probabilities (simplified)
emit_prob = {
    'B': {'自': 0.5, '清': 0.5},
    'M': {'然': 0.5, '华': 0.5},
    'E': {'言': 0.5, '学': 0.5},
    'S': {'我': 0.5, '来': 0.5}
}


def viterbi(sentence):

    V = [{}]
    path = {}

    # Initialization
    for y in states:
        emit = emit_prob.get(y, {}).get(sentence[0], 1e-6)
        V[0][y] = start_prob.get(y, 0) * emit
        path[y] = [y]

    # Dynamic programming
    for t in range(1, len(sentence)):
        V.append({})
        new_path = {}

        for y in states:

            emit = emit_prob.get(y, {}).get(sentence[t], 1e-6)

            prob, state = max(
                (
                    V[t - 1][y0]
                    * trans_prob.get(y0, {}).get(y, 1e-6)
                    * emit,
                    y0
                )
                for y0 in states
            )

            V[t][y] = prob
            new_path[y] = path[state] + [y]

        path = new_path

    # Final state
    n = len(sentence) - 1

    prob, state = max((V[n][y], y) for y in states)

    return path[state]


def segment(sentence, tags):

    result = []
    word = ""

    for char, tag in zip(sentence, tags):

        if tag == 'S':
            result.append(char)

        elif tag == 'B':
            word = char

        elif tag == 'M':
            word += char

        elif tag == 'E':
            word += char
            result.append(word)

    return result


sentence = "自然语言"

tags = viterbi(sentence)

result = segment(sentence, tags)

print("Sentence:")
print(sentence)

print("\nTags:")
print(tags)

print("\nSegmentation:")
print(result)
```

---

## Example Output

```Python
Sentence:
自然语言

Tags:
['B', 'M', 'M', 'E']

Segmentation:
['自然语言']
```

## N‑Gram Word Segmentation and Probability Calculation

```Python
from collections import defaultdict

corpus = ["我爱自然语言处理", "自然语言处理很有趣"]
# generate all 生成所有N-Gram并统计频率
def generate_n_grams(texts, n):
    n_gram_counts = defaultdict(int)
    for text in texts:
        words = list(text)  # 假设已分词为字符级
        for i in range(len(words) - n + 1):
            n_gram = ''.join(words[i:i+n])
            n_gram_counts[n_gram] += 1
    return n_gram_counts
# compute 
def calculate_probability(n_gram_counts, n):
    probabilities = {}
    for n_gram, count in n_gram_counts.items():
        if n == 1:
            probabilities[n_gram] = count / sum(n_gram_counts.values())
        else:
            prefix = n_gram[:-1]  # 获取前N-1个字符
            prefix_count = sum(g for g in n_gram_counts if g.startswith(prefix[:-1]) if len(g) == n-1)  # 简化处理，实际需更精确
            # 更精确的实现需构建前缀树或使用滑动窗口统计
            probabilities[n_gram] = count / (prefix_count if prefix_count > 0 else 1)  # 避免除零
    return probabilities
# 生成2-Gram并计算概率
n = 2
n_gram_counts = generate_n_grams(corpus, n)
probabilities = calculate_probability(n_gram_counts, n)
print("2-Gram频率统计:", n_gram_counts)
print("2-Gram条件概率:", probabilities)
```
